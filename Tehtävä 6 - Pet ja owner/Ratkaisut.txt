Käytetyt komennot:
CREATE TABLE owner (ownerID int PRIMARY KEY AUTO_INCREMENT, first_name varchar(255), last_name varchar(255), address varchar(255));
ALTER TABLE pet ADD COLUMN petID INT PRIMARY KEY AUTO_INCREMENT;
ALTER TABLE pet ADD COLUMN ownerID INT;
ALTER TABLE pet ADD CONSTRAINT FOREIGN KEY(ownerID) REFERENCES owner(ownerID);

Koska owner-taulun rakenne on eri kuin esimerkissä, pitää NULL olla eri paikassa kuin esimerkissä:
INSERT INTO owner VALUES(NULL, 'Harold', 'Lloyd', 'Someaddress 1');
INSERT INTO owner VALUES(NULL, 'Gwen', 'Stefani', 'Someaddress 2');
INSERT INTO owner VALUES(NULL, 'Benny', 'Hill', 'Someaddress 3');
INSERT INTO owner VALUES(NULL, 'Diane', 'Keaton', 'Someaddress 4');

Koska owner-saraketta ei ole vielä poistettu pet-taulusta, voidaan yhdistää ownerID owner-sarakkeen arvon avulla:
UPDATE pet SET ownerID = (SELECT ownerID FROM owner WHERE first_name = pet.owner);

Nyt voidaan poistaa owner-sarake pet-taulusta:
ALTER TABLE pet DROP column owner;
